\chapter{Prerequisites}

\noindent Arfken,\cite{arfken2012} Spivak,\cite{spivak1965} and Baez and
Muniain\cite{baez1994gauge} are good sources for background on \textit{groups},
\textit{vector spaces} and \textit{differentiable manifolds}. 

We adhere to the following terse definitions in the rest of these notes.

\begin{definition}[Lie Algebra]
    A Lie algebra is a vector space $\mathfrak{g}$ equipped with a bilinear,
    anticommutative, binary operation
    $[,]:\mathfrak{g}\times\mathfrak{g}\rightarrow\mathfrak{g}$ that satisfies
    the Jacobi identity:
    \begin{equation*}
        [X,[Y,Z]]+[Z,[X,Y]+[Y,[Z,X]]=0
    \end{equation*}
\end{definition}

\begin{definition}[Lie Group]
    A Lie group is a group and also a differentiable manifold.
\end{definition}

\begin{remark}
A Lie algebra is not a group under the Lie bracket and is not a differentiable
manifold.
\end{remark}
