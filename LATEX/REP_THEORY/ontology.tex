\section{Relationships}

\begin{fullwidth}
    \centering
    \captionof{figure}{Ontology of Objects and Mappings}\label{fig:bridge_ontology}
    \vspace{10pt}
    \begin{tikzpicture}[
        scale=0.85, every node/.style={scale=0.85},
        node distance=1.2cm and 0.8cm,
        block/.style={rectangle, draw, rounded corners, minimum width=2.8cm, minimum height=0.8cm, align=center, fill=gray!5},
        action/.style={rectangle, draw, dashed, rounded corners, minimum width=2.8cm, minimum height=0.8cm, align=center, fill=blue!5},
        lbl/.style={fill=white, fill opacity=0.85, text opacity=1, inner sep=1.5pt, rounded corners=1pt},
        arrow/.style={-Stealth, thick},
        bij/.style={Stealth-Stealth, thick},
        line/.style={thick}
    ]

    % Pillars: Left (Spin/Complex) and Right (Vector/Real)
    
    % Level 1: Core Algebra
    \node[block] (g) {Abstract $\mathfrak{su}(2)$};
    \node[block] (f) [above=1.2cm of g] {$f_{ijk}$ Constants};

    % Level 2: Matrix Representations
    \node[block] (su2) [below left=1.5cm and 1.5cm of g] {$\mathfrak{su}(2)$ 2x2 ($T$)};
    \node[block] (so3) [below right=1.5cm and 1.5cm of g] {$\mathfrak{so}(3)$ 3x3 ($J$)};

    % Level 3: Groups
    \node[block] (SU2) [below=2.0cm of su2] {$SU(2)$ Group};
    \node[block] (SO3) [below=2.0cm of so3] {$SO(3)$ Group};
    \node[block] (Center) [left=0.8cm of SU2] {Center $\{I, -I\}$};

    % Level 4: Final Objects (The "Wings" Layout)
    \node[block] (Spin) [below=2.0cm of SU2] {Spinors ($\mathbb{C}^2$)};
    \node[block] (R3) [below=2.0cm of SO3] {Real $\mathbb{R}^3$};
    
    % Centered between the pillars
    \node[block] (h0) at ($(Spin)!0.5!(R3)$) {Mat-Vecs ($\mathfrak{h}_0$)};
    
    % Far outer wings for topology
    \node[block] (S3) [left=0.8cm of Spin] {$S^3$ Sphere};
    \node[block] (RP3) [right=0.8cm of R3] {$\mathbb{RP}^3$ Space};

    % --- EDGES ---

    % Algebra
    \draw[bij] (f) -- node[lbl, right] {\footnotesize Bij (Basis)} (g);
    \draw[line] (g) -- (su2);
    \draw[line] (g) -- (so3);
    \draw[bij] (su2) -- node[lbl, above] {\footnotesize Bij: TtoJ} (so3);

    % Groups
    \draw[arrow] (su2) -- node[lbl, left] {\footnotesize Surj: Exp} (SU2);
    \draw[arrow] (so3) -- node[lbl, right] {\footnotesize Surj: Exp} (SO3);
    \draw[arrow] (Center) -- node[lbl, above] {\footnotesize Inj} (SU2);
    \draw[arrow] (SU2) -- node[lbl, above] {\footnotesize Surj (2:1)} (SO3);

    % Topology (The outer-wing paths)
    \draw[bij] (SU2) -- node[lbl, midway, above left] {\footnotesize Bij (Diffeo)} (S3);
    \draw[bij] (SO3) -- node[lbl, midway, above right] {\footnotesize Bij (Diffeo)} (RP3);

    % Actions
    \draw[arrow, dashed] (SU2) -- node[lbl, midway, left] {\footnotesize Act: Fund} (Spin);
    \draw[arrow, dashed] (SU2) -- node[lbl, midway, above right, xshift=-5pt] {\footnotesize Act: Adjoint} (h0);
    \draw[arrow, dashed] (SO3) -- node[lbl, midway, left] {\footnotesize Act: Rotation} (R3);

    % Vector Space Isometry
    \draw[bij] (h0) -- node[lbl, midway, below=12pt] {\footnotesize Bij: Pauli Map} (R3);

    \end{tikzpicture}
